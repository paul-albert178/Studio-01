PRAGMA foreign_keys = ON;

CREATE TABLE Companies (
    company_id  INTEGER PRIMARY KEY AUTOINCREMENT,
    name        TEXT NOT NULL,
    email       TEXT,
    phone       TEXT,
    address     TEXT
);

CREATE TABLE Job_Role (
    job_id             INTEGER PRIMARY KEY AUTOINCREMENT,
    company_id         INTEGER NOT NULL,
    job_title          TEXT NOT NULL,
    department         TEXT,
    location           TEXT,
    experience_level   TEXT,
    status             TEXT,
    FOREIGN KEY (company_id) REFERENCES Companies(company_id) ON DELETE CASCADE
);

CREATE TABLE Applicants (
    applicant_id  INTEGER PRIMARY KEY AUTOINCREMENT,
    job_id        INTEGER NOT NULL,
    full_name     TEXT NOT NULL,
    email         TEXT,
    phone         TEXT,
    resume        TEXT,
    apply_date    DATE,
    status        TEXT,
    FOREIGN KEY (job_id) REFERENCES Job_Role(job_id) ON DELETE CASCADE
);

CREATE TABLE Interview_Panel (
    interviewer_id  INTEGER PRIMARY KEY AUTOINCREMENT,
    name            TEXT NOT NULL,
    email           TEXT,
    role            TEXT,
    expertise       TEXT
);

CREATE TABLE Interviews (
    interview_id     INTEGER PRIMARY KEY AUTOINCREMENT,
    applicant_id     INTEGER NOT NULL,
    job_id           INTEGER NOT NULL,
    interviewer_id   INTEGER NOT NULL,
    interview_date   DATETIME,
    type             TEXT CHECK (type IN ('AI', 'Video', 'Live')),
    status           TEXT,
    FOREIGN KEY (applicant_id)   REFERENCES Applicants(applicant_id) ON DELETE CASCADE,
    FOREIGN KEY (job_id)         REFERENCES Job_Role(job_id) ON DELETE CASCADE,
    FOREIGN KEY (interviewer_id) REFERENCES Interview_Panel(interviewer_id)
);

CREATE TABLE Feedback (
    feedback_id     INTEGER PRIMARY KEY AUTOINCREMENT,
    interview_id    INTEGER NOT NULL,
    interviewer_id  INTEGER NOT NULL,
    rating          INTEGER,
    comments        TEXT,
    ai_score        REAL,
    feedback_date   DATETIME,
    FOREIGN KEY (interview_id)   REFERENCES Interviews(interview_id) ON DELETE CASCADE,
    FOREIGN KEY (interviewer_id) REFERENCES Interview_Panel(interviewer_id)
);

CREATE TABLE Offers (
    offer_id      INTEGER PRIMARY KEY AUTOINCREMENT,
    applicant_id  INTEGER NOT NULL,
    job_id        INTEGER NOT NULL,
    offer_date    DATE,
    salary        REAL,
    status        TEXT,
    remarks       TEXT,
    FOREIGN KEY (applicant_id) REFERENCES Applicants(applicant_id) ON DELETE CASCADE,
    FOREIGN KEY (job_id)       REFERENCES Job_Role(job_id) ON DELETE CASCADE
);
